def levenshtein(a, b):
    """Edit distance between two strings, O(len(a) * len(b)) time, O(len(b)) space."""
    if a == b:
        return 0
    if not a:
        return len(b)
    if not b:
        return len(a)

    previous = list(range(len(b) + 1))
    for i, ca in enumerate(a, start=1):
        current = [i]
        for j, cb in enumerate(b, start=1):
            insert = current[j - 1] + 1
            delete = previous[j] + 1
            substitute = previous[j - 1] + (ca != cb)
            current.append(min(insert, delete, substitute))
        previous = current
    return previous[-1]

import random


def reservoir_sample(stream, k, rng=None):
    """Uniformly sample k items from a stream of unknown length in one pass.

    Every item in the stream ends up in the reservoir with probability k/n,
    where n is the final length, without n ever being known in advance.
    """
    rng = rng or random.Random()
    reservoir = []
    for index, item in enumerate(stream):
        if index < k:
            reservoir.append(item)
        else:
            j = rng.randint(0, index)
            if j < k:
                reservoir[j] = item
    return reservoir
